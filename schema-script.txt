-- ============================================================
-- NOTE: This file is the legacy MySQL schema and is not for Cloudflare D1.
-- Use schema-d1.sql for the single Cloudflare SQLite database.
-- ============================================================
--  TaskFlow MySQL Schema
--  Engine: InnoDB  |  Charset: utf8mb4  |  Collation: utf8mb4_unicode_ci
--
--  Design notes
--  ------------
--  • Username is the sole user identifier (max 14 chars, case-insensitive,
--    unique).  Passwords are stored as SHA-256 hex strings - sufficient for
--    low-sensitivity data as discussed.
--  • Every mutable row carries an `updated_at` TIMESTAMP(3) (millisecond
--    precision) for last-write-wins conflict resolution.  When the same
--    item is edited on two devices simultaneously the server keeps whichever
--    update arrives with the later `updated_at`.
--  • Soft-delete is NOT used.  Rows are hard-deleted.  The server-side
--    cleanup event enforces each user's `delete_after_days` setting nightly.
--  • Recurrence data and transition sub-settings are stored as JSON columns
--    to avoid extra normalisation overhead for small, structured blobs.
--  • No sessions table - web clients send credentials on every page load;
--    mobile clients stay authenticated via an in-memory token valid for the
--    lifetime of the app process (no persistent session row needed).
-- ============================================================

-- ─── 0. DATABASE ──────────────────────────────────────────────────────────────

CREATE DATABASE IF NOT EXISTS taskflow
  CHARACTER SET utf8mb4
  COLLATE utf8mb4_unicode_ci;

USE taskflow;

-- ─── 1. USERS ─────────────────────────────────────────────────────────────────

CREATE TABLE IF NOT EXISTS users (
  -- Identity
  username          VARCHAR(14)   NOT NULL,
  password_hash     CHAR(64)      NOT NULL,        -- SHA-256 hex, lowercase

  -- Timestamps
  created_at        TIMESTAMP(3)  NOT NULL DEFAULT CURRENT_TIMESTAMP(3),
  updated_at        TIMESTAMP(3)  NOT NULL DEFAULT CURRENT_TIMESTAMP(3)
                                  ON UPDATE CURRENT_TIMESTAMP(3),

  PRIMARY KEY (username)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

-- Username is inherently unique via PK; enforce case-insensitivity through
-- the utf8mb4_unicode_ci collation (a = A).

-- ─── 2. USER SETTINGS ─────────────────────────────────────────────────────────
-- One row per user.  All app preferences live here.
-- Mirrors every localStorage key except items/categories/stats.

CREATE TABLE IF NOT EXISTS user_settings (
  username                  VARCHAR(14)   NOT NULL,

  -- Theme / appearance
  light_color               CHAR(7)       NOT NULL DEFAULT '#344271',  -- hex
  dark_color                CHAR(7)       NULL DEFAULT NULL,            -- hex, null = use light_color
  color_palette             JSON          NOT NULL,                     -- array of hex strings
  is_dark_mode              TINYINT(1)    NOT NULL DEFAULT 0,
  is_dim_mode               TINYINT(1)    NOT NULL DEFAULT 0,           -- legacy: 1 = "lighter dark" variant of dark mode
  theme_level               TINYINT       NOT NULL DEFAULT 0,           -- 0=full dark … 4=full light (0-2 dark family)
  week_calendar_days        TINYINT       NOT NULL DEFAULT 7,             -- 5|7 days shown in the week calendar

  -- Animations
  enable_animations         TINYINT(1)    NOT NULL DEFAULT 1,
  confetti_enabled          TINYINT(1)    NOT NULL DEFAULT 1,
  completion_sounds         TINYINT(1)    NOT NULL DEFAULT 1,
  popup_overlay_enabled     TINYINT(1)    NOT NULL DEFAULT 1,
  page_transitions_enabled  TINYINT(1)    NOT NULL DEFAULT 1,
  smooth_dropdowns_enabled  TINYINT(1)    NOT NULL DEFAULT 1,
  deletion_fade_enabled     TINYINT(1)    NOT NULL DEFAULT 1,
  split_view_shift_enabled  TINYINT(1)    NOT NULL DEFAULT 1,
  button_feedback_enabled   TINYINT(1)    NOT NULL DEFAULT 1,
  split_task_page           TINYINT(1)    NOT NULL DEFAULT 1,
  calendar_days             TINYINT       NOT NULL DEFAULT 1,             -- 1|3|5|7 days shown in standalone calendar

  -- Calendar / timeline
  week_start                TINYINT       NOT NULL DEFAULT 0,           -- 0=Sunday, 1=Monday
  day_start                 CHAR(5)       NOT NULL DEFAULT '07:00',     -- HH:MM
  day_end                   CHAR(5)       NOT NULL DEFAULT '20:00',     -- HH:MM
  timeline_increment        SMALLINT      NOT NULL DEFAULT 15,          -- minutes: 15|30|60
  task_creation_increment   SMALLINT      NOT NULL DEFAULT 15,          -- minutes: 5|10|15
  future_recurrences        SMALLINT      NOT NULL DEFAULT 2,           -- upcoming occurrences kept per recurring series: 1|2
  use_military_time         TINYINT(1)    NOT NULL DEFAULT 0,

  -- Notifications (global, not per-item)
  task_notifications        JSON          NOT NULL,
  -- task_notifications shape: [{"id":1,"offset":0},{"id":2,"offset":5}]
  --   offset = minutes before deadline to fire notification
  event_notifications       JSON          NOT NULL,
  -- same shape as task_notifications

  -- UI
  quick_add_only            TINYINT(1)    NOT NULL DEFAULT 0,
  split_ratio               FLOAT         NOT NULL DEFAULT 0.5,         -- 0.25–0.75

  -- Data retention
  delete_after_days         SMALLINT      NOT NULL DEFAULT 30,
  -- Allowed values: 14 (2 weeks), 30 (1 month), 90 (3 months), 0 = never
  task_deadline_push_enabled  TINYINT(1)   NOT NULL DEFAULT 1,
  -- Settings > General > "Push back past due date": if 1, incomplete tasks keep
  -- rolling to the next day at midnight even past their due date; if 0 they roll
  -- only until they reach the due date and then park there (the app flags them
  -- red as priority). The column name predates the due-date feature.
  -- App-wide typeface label from Settings > General > App font: an
  -- allow-listed name like 'Inter'/'Times New Roman' or the sentinel 'Default'
  -- (see Split Task/src/fonts.js). VARCHAR(32) fits the longest label.
  app_font                  VARCHAR(32)   NOT NULL DEFAULT 'Inter',

  -- Sync
  updated_at                TIMESTAMP(3)  NOT NULL DEFAULT CURRENT_TIMESTAMP(3)
                                          ON UPDATE CURRENT_TIMESTAMP(3),

  PRIMARY KEY (username),
  CONSTRAINT fk_settings_user
    FOREIGN KEY (username) REFERENCES users (username)
    ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

-- ─── 3. CATEGORIES ────────────────────────────────────────────────────────────

CREATE TABLE IF NOT EXISTS categories (
  id          VARCHAR(32)   NOT NULL,              -- 'default' or 'cat_N'
  username    VARCHAR(14)   NOT NULL,
  name        VARCHAR(64)   NOT NULL,
  color       CHAR(7)       NOT NULL,              -- hex e.g. '#4A6CF7'
  updated_at  TIMESTAMP(3)  NOT NULL DEFAULT CURRENT_TIMESTAMP(3)
                            ON UPDATE CURRENT_TIMESTAMP(3),

  PRIMARY KEY (id, username),
  CONSTRAINT fk_categories_user
    FOREIGN KEY (username) REFERENCES users (username)
    ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

-- ─── 4. ITEMS (TASKS & EVENTS) ────────────────────────────────────────────────

CREATE TABLE IF NOT EXISTS items (
  id                VARCHAR(32)   NOT NULL,        -- client-generated, e.g. Date.now().toString()
  username          VARCHAR(14)   NOT NULL,

  -- Core fields
  type              ENUM('task','event')  NOT NULL DEFAULT 'task',
  title             VARCHAR(255)  NOT NULL,
  category_id       VARCHAR(32)   NOT NULL DEFAULT 'default',
  item_date         DATE          NOT NULL,        -- the calendar day
  done              TINYINT(1)    NOT NULL DEFAULT 0,
  finished_at       TIMESTAMP(3)  NULL,            -- when the item was completed (drives "finished" ordering)
  notes             TEXT          NULL,

  -- Task-only: the day the task must be done by + the time on that day.
  -- due_date is NULL on legacy rows (and on quick-added tasks), which carry
  -- deadline - the time - only.
  deadline          VARCHAR(20)   NULL,            -- e.g. '09:30 AM'
  due_date          DATE          NULL,            -- 'YYYY-MM-DD'; NULL = no due date

  -- Event-only
  start_time        VARCHAR(20)   NULL,
  end_time          VARCHAR(20)   NULL,
  location          VARCHAR(255)  NULL,

  -- Recurrence (stored as JSON for flexibility)
  recurrence        JSON          NOT NULL,
  -- recurrence shape:
  -- {
  --   "type":     "none"|"daily"|"weekly"|"month"|"year"|"custom",
  --   "days":     [],       -- array of weekday ints (0–6)
  --   "interval": 1         -- used for "custom" type
  -- }
  recurrence_id     VARCHAR(32)   NULL,            -- links generated copies to canonical item

  -- Conflict resolution
  updated_at        TIMESTAMP(3)  NOT NULL DEFAULT CURRENT_TIMESTAMP(3)
                                  ON UPDATE CURRENT_TIMESTAMP(3),

  PRIMARY KEY (id, username),
  INDEX idx_items_username_date (username, item_date),
  INDEX idx_items_recurrence_id (recurrence_id),
  CONSTRAINT fk_items_user
    FOREIGN KEY (username) REFERENCES users (username)
    ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

-- ─── 5. SERVER-SIDE DATA CLEANUP EVENT ────────────────────────────────────────
-- Runs nightly at 02:00 server time.
-- Deletes items older than each user's delete_after_days setting.
-- Users with delete_after_days = 0 are skipped (keep forever).

DELIMITER $$

CREATE EVENT IF NOT EXISTS ev_purge_old_items
  ON SCHEDULE EVERY 1 DAY
  STARTS (DATE(NOW()) + INTERVAL 1 DAY + INTERVAL 2 HOUR)
  DO
  BEGIN
    DELETE items
    FROM   items
    JOIN   user_settings AS s ON items.username = s.username
    WHERE  s.delete_after_days > 0
      AND  items.item_date < (CURDATE() - INTERVAL s.delete_after_days DAY);
  END$$

DELIMITER ;

-- ─── 6c. MIGRATION NOTE (login rate limiting) ─────────────────────────────────
-- The worker enforces brute-force protection on /api/login using a small
-- counter table. It is created LAZILY by the worker on the first login request
-- (CREATE TABLE IF NOT EXISTS), so no manual step is required for D1. For
-- reference / manual setup, the D1 DDL is:
--
--   CREATE TABLE IF NOT EXISTS login_attempts (
--     key TEXT PRIMARY KEY,        -- e.g. 'user:alice' or 'ip:203.0.113.9'
--     count INTEGER NOT NULL DEFAULT 0,
--     window_start INTEGER NOT NULL -- epoch ms; counters reset when stale
--   );
--
-- After 5 failed attempts within 15 minutes (per account OR per IP) the login
-- endpoint returns HTTP 429. A successful login clears the account counter.
-- Expired rows are pruned nightly by the worker's scheduled() cron.

-- ─── 6d. MIGRATION NOTE (conditional pull / user_versions) ───────────────────
-- GET /api/sync takes an optional `version` parameter. When it matches the
-- caller's row in `user_versions`, the worker answers `{ unchanged: true }`
-- WITHOUT reading items/categories/settings - so an idle poll costs two tiny
-- lookups instead of three result sets plus the JSON round trip. Every
-- POST /api/sync write bumps that row by one, which is also what tells a
-- user's other devices to re-fetch. The table is created lazily by the worker
-- (ensureSchema on the pull path), so no manual step is required for D1:
--
--   CREATE TABLE IF NOT EXISTS user_versions (
--     username TEXT NOT NULL PRIMARY KEY,   -- FK → users(username) ON DELETE CASCADE
--     version  INTEGER NOT NULL DEFAULT 0   -- bumped by every sync write
--   );
--
-- `ensureSchema` also creates the username indexes (idx_items_username,
-- idx_items_username_date, idx_items_recurrence_id, idx_categories_username)
-- lazily, so a database created before they existed stops full-scanning
-- items/categories on every pull.

-- ─── 8. CONFLICT RESOLUTION LOGIC (app-server contract) ──────────────────────
-- The API endpoint for updating any row should use INSERT ... ON DUPLICATE KEY
-- UPDATE with a check on updated_at, e.g. for items:
--
--   INSERT INTO items (id, username, type, title, ..., updated_at)
--   VALUES (?, ?, ?, ?, ..., ?)
--   ON DUPLICATE KEY UPDATE
--     title        = IF(VALUES(updated_at) >= updated_at, VALUES(title),        title),
--     done         = IF(VALUES(updated_at) >= updated_at, VALUES(done),         done),
--     notes        = IF(VALUES(updated_at) >= updated_at, VALUES(notes),        notes),
--     deadline     = IF(VALUES(updated_at) >= updated_at, VALUES(deadline),     deadline),
--     due_date     = IF(VALUES(updated_at) >= updated_at, VALUES(due_date),     due_date),
--     start_time   = IF(VALUES(updated_at) >= updated_at, VALUES(start_time),   start_time),
--     end_time     = IF(VALUES(updated_at) >= updated_at, VALUES(end_time),     end_time),
--     location     = IF(VALUES(updated_at) >= updated_at, VALUES(location),     location),
--     recurrence   = IF(VALUES(updated_at) >= updated_at, VALUES(recurrence),   recurrence),
--     category_id  = IF(VALUES(updated_at) >= updated_at, VALUES(category_id),  category_id),
--     item_date    = IF(VALUES(updated_at) >= updated_at, VALUES(item_date),     item_date),
--     updated_at   = IF(VALUES(updated_at) >= updated_at, VALUES(updated_at),   updated_at);
--
-- Same pattern applies to categories and user_settings rows.
-- "Last timestamp wins" - no field-level merging, entire row goes to winner.

-- ─── 9. SESSION / AUTH CONTRACT (no sessions table needed) ───────────────────
-- Web:   Client POSTs username+password on every page load.
--        Server verifies SHA-256(password) == password_hash.
--        Server returns a short-lived signed token (e.g. HMAC-SHA256 JWT,
--        valid for the browser tab's lifetime - no expiry longer than session).
--        Token is stored in sessionStorage (NOT localStorage) so it is cleared
--        on tab close / reload.
--
-- Mobile (Capacitor): App sends credentials once on first launch or after
--        explicit logout.  Token is stored in device memory / secure storage
--        for the lifetime of the app process.  No row in the DB needed.
--
-- Both: Every API request carries the token in the Authorization header.
--       Server validates the token on each request without a DB round-trip.
